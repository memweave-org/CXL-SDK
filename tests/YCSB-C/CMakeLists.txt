macro(add_db db_file db_dir lib_name program_macro link_type)
    list(APPEND SUBSRCS db/${db_file})
    if(${link_type} STREQUAL "PUBLIC")
        list(APPEND public_link_libs ${lib_name})
    elseif(${link_type} STREQUAL "INTERFACE")
        list(APPEND interface_link_libs ${lib_name})
    elseif(${link_type} STREQUAL "HEADER")
        list(APPEND header_link_libs ${lib_name})
    endif()
    if(${link_type} STREQUAL "PUBLIC" OR ${link_type} STREQUAL "INTERFACE")
        add_subdirectory(../../ds/${db_dir} ${CMAKE_BINARY_DIR}/${db_dir})
    else()
        include_directories(../../ds/${db_dir})
    endif()
    add_compile_definitions(${program_macro})
endmacro()

# Define the executable
macro(add_ycsbc_executable target_name compile_definitions)
    set(nt_compile_definitions ${compile_definitions})
    if(NOT UB_API STREQUAL "none")
        list(REMOVE_ITEM nt_compile_definitions QUEUE_SIM_CAS NT_SIM)
    endif()
    string(REPLACE "ycsbc_" "" short_target_name ${target_name})
    string(TOUPPER ${short_target_name} upper_target_name)
    if(VARIANT STREQUAL ${short_target_name})
        message(STATUS "DEF: ${nt_compile_definitions}")
        add_compile_definitions(${nt_compile_definitions})
        add_executable(${target_name} ${SUBSRCS})
        if(YCSB_UB_LOCAL_NCC)
            local_ncc_instrument(${target_name})
            if(TARGET bwtree)
                local_ncc_instrument(bwtree)
            endif()
        endif()
        target_include_directories(${target_name} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
        # Create unique library targets for each executable
        foreach(lib ${public_link_libs})
            target_link_libraries(${target_name} ${lib})
            get_target_property(lib_type ${lib} TYPE)
            if(lib_type STREQUAL "INTERFACE_LIBRARY")
                target_compile_definitions(${lib} INTERFACE ${nt_compile_definitions})
            else()
                target_compile_definitions(${lib} PUBLIC ${nt_compile_definitions})
            endif()
        endforeach()
        foreach(lib ${interface_link_libs})
            target_link_libraries(${target_name} ${lib})
            target_compile_definitions(${lib} INTERFACE ${nt_compile_definitions})
        endforeach()
        # 查找并链接系统库，避免依赖 RadixART 才带上它们
        find_library(JemallocLib jemalloc)
        if(JemallocLib)
            target_link_libraries(${target_name} ${JemallocLib})
        endif()
        target_link_libraries(${target_name} pthread tbb atomic ssh numa memkind)
        option(YCSB_BUILD_READONLY_RPC_CONTRACT "Build untimed A-read-only and shared-queue assertions (fixture SDK only)" OFF)
        if(YCSB_BUILD_READONLY_RPC_CONTRACT)
            if(NOT YCSB_UB_TWO_HOST)
                message(FATAL_ERROR "Read-only RPC contract requires YCSB_UB_TWO_HOST")
            endif()
            set(contract_sources ${SUBSRCS})
            list(FILTER contract_sources EXCLUDE REGEX "/ycsbc[.]cc$")
            add_executable(readonly_rpc_contract ub/readonly_rpc_contract.cc ${contract_sources})
            add_executable(rpc_queue_router_contract ub/rpc_queue_router_contract.cc)
            target_link_libraries(rpc_queue_router_contract PRIVATE pthread)
            get_target_property(contract_libraries ${target_name} LINK_LIBRARIES)
            target_link_libraries(readonly_rpc_contract PRIVATE ${contract_libraries})
            if("clevel-hash" IN_LIST YCSB_INDEXES)
                add_executable(ub_reuse_contract ub/ub_reuse_contract.cc)
                target_link_libraries(ub_reuse_contract PRIVATE ${contract_libraries})
                if(UB_API STREQUAL "ubsm")
                    add_executable(clevel_level_gc_contract ub/clevel_level_gc_contract.cc)
                    target_link_libraries(clevel_level_gc_contract PRIVATE ${contract_libraries})
                endif()
            endif()
        endif()
        set_target_properties(${target_name} PROPERTIES ENABLE_EXPORTS ON)
        if(NOT YCSB_UB_TWO_HOST)
            set_target_properties(${target_name} PROPERTIES POSITION_INDEPENDENT_CODE TRUE)
        endif()
    endif()
endmacro()
