cmake_minimum_required(VERSION 3.16)
project(two_process_ycsb LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(SYSTEM bwtree CACHE STRING "sherman, rpc, bwtree, clevelhash")
option(OPT "Paper SDK optimizations" ON)
option(CC "Hardware cache coherence (without NO_CC or OPT macros)" OFF)
set(BWTREE_MAPPING_TABLE_BITS 24 CACHE STRING "24 for paper scale")
add_compile_options(-O3 -g -march=native -UNDEBUG)
add_compile_definitions(SHM_DISABLE_CAS128 TWO_PROCESS_SHARED_COPY BWTREE_MAPPING_CACHE_DRAM=0 BWTREE_MAPPING_TABLE_BITS=${BWTREE_MAPPING_TABLE_BITS})
if(NOT CC)
  add_compile_definitions(NO_CC QUEUE_SIM_CAS)
endif()
if(OPT AND NOT CC)
  add_compile_definitions(OPT_GC OPT_ROOT_READ OPT_IN_USE_FLAG OPT_REPLICATE_META OPT_BATCH_FLUSH OPT_CLEVEL_ROOT_READ)
endif()
add_subdirectory(../../shm-lib shm-lib)
add_library(transport ../../ds/cxl_transport/memory.cc)
target_link_libraries(transport PUBLIC shm-lib atomic)
set(SOURCES main.cc ../YCSB-C/core/core_workload.cc ../YCSB-C/core/utils.cc)
if(SYSTEM STREQUAL "deft")
  message(FATAL_ERROR "Deft is excluded: UB does not support its required 128-bit CAS")
elseif(SYSTEM STREQUAL "sherman")
  list(APPEND SOURCES sherman.cc ../../ds/Sherman/src/Tree.cpp ../../ds/Sherman/src/DSM.cpp ../../ds/Sherman/src/Cache.cpp ../../ds/Sherman/src/GlobalLock.cpp ../../ds/Sherman/src/Debug.cpp)
  set(INDEX_INCLUDE ../../ds/Sherman/include)
  add_compile_definitions(SHERMAN_PAGE_SIZE=1024)
elseif(SYSTEM STREQUAL "bwtree" OR SYSTEM STREQUAL "rpc")
  list(APPEND SOURCES bwtree.cc ../../ds/BwTree/src/bwtree.cpp ../../ds/BwTree/src/test_suite.cpp)
  set(INDEX_INCLUDE ../../ds/BwTree/src)
  if(SYSTEM STREQUAL "rpc")
    add_compile_definitions(TWO_PROCESS_RPC USE_NO_CC_QUEUE)
  endif()
elseif(SYSTEM STREQUAL "clevelhash")
  list(APPEND SOURCES clevelhash.cc)
  set(INDEX_INCLUDE ../../ds/ClevelHash)
else()
  message(FATAL_ERROR "Unknown SYSTEM")
endif()
add_executable(bench ${SOURCES})
target_include_directories(bench PRIVATE ${INDEX_INCLUDE} ../YCSB-C/include)
target_link_libraries(bench PRIVATE transport pthread numa memkind atomic cityhash boost_coroutine boost_context)
target_compile_definitions(bench PRIVATE BENCH_SYSTEM="${SYSTEM}" BENCH_OPT=$<AND:$<BOOL:${OPT}>,$<NOT:$<BOOL:${CC}>>> BENCH_CC=$<BOOL:${CC}>)
